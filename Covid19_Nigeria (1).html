state = (
    df.groupby('Location')[['Nr. Of New Cases', 'Total Confirmed Cases',
                             'Deaths', 'Recovered']]
    .max()
    .reset_index()
    .sort_values('Total Confirmed Cases', ascending=False)
)

fig, axes = plt.subplots(1, 2, figsize=(16, 7))
fig.suptitle('State-Level COVID-19 Breakdown', fontsize=15, fontweight='bold')

top_states = state.nlargest(15, 'Total Confirmed Cases')
colors = [RED if v == top_states['Total Confirmed Cases'].max()
          else BLUE for v in top_states['Total Confirmed Cases']]
axes[0].barh(top_states['Location'], top_states['Total Confirmed Cases'],
             color=colors)
axes[0].invert_yaxis()
axes[0].set_title('Top 15 States by Total Confirmed Cases')
axes[0].set_xlabel('Total Confirmed Cases')
for i, v in enumerate(top_states['Total Confirmed Cases']):
    axes[0].text(v + 0.3, i, str(v), va='center', fontsize=9)

deaths_df = state[state['Deaths'] > 0].sort_values('Deaths', ascending=False)
axes[1].barh(deaths_df['Location'], deaths_df['Deaths'], color=RED)
axes[1].invert_yaxis()
axes[1].set_title('Deaths by State')
axes[1].set_xlabel('Deaths')
